function esc(s) {
  var a = String.fromCharCode(38);
  return String(s || '').replace(/&/g, a + 'amp;').replace(/</g, a + 'lt;').replace(/>/g, a + 'gt;').replace(/"/g, a + 'quot;');
}

function renderForum() {
  var el = document.getElementById('forumRawData');
  var raw = el ? el.value : '';
  var lines = raw.split('\n');
  var posts = [], map = {};

  for (var i = 0; i < lines.length; i++) {
    var s = lines[i].trim();
    var left = s.indexOf('[');
    var right = s.lastIndexOf(']');
    if (left === -1 || right <= left) continue;

    var inside = s.substring(left + 1, right).replace(/｜/g, '|');
    var parts = inside.split('|');
    for (var j = 0; j < parts.length; j++) parts[j] = parts[j].trim();

    if (parts[0] === '帖' && parts.length >= 5) {
      var id = parts[1] || '?';
      var cleanId = id.replace(/^[#＃]+/, '');
      var author = parts[2] || '匿名';
      var time = parts[3] || '刚刚';
      var title = parts[4] || '无题';
      var heat = (parts.length >= 7) ? parts[5] : '';
      var body = parts[parts.length - 1] || '';
      var p = { id: id, cleanId: cleanId, author: author, time: time, title: title, heat: heat, body: body, reps: [] };
      posts.push(p);
      map[cleanId] = p;
    } else if (parts[0] === '回复' && parts.length >= 3) {
      var rawTarget = parts[1] || '';
      var cleanTarget = rawTarget.replace(/^[#＃]+/, '');
      var parentId = cleanTarget.split('-')[0].trim();
      var repAuthor = parts[2] || '匿名';
      var repTime = (parts.length >= 5) ? parts[3] : '';
      var repText = parts[parts.length - 1] || '';
      if (map[parentId]) {
        map[parentId].reps.push({ author: repAuthor, time: repTime, text: repText });
      }
    }
  }

  var h = '<div class="t-top">' +
    '<div class="t-title">📡 全球绿皮火车 · 跨界匿名终端 <span class="t-badge">车厢屏</span></div>' +
    '<div class="t-hint">点击标题展开讨论</div></div>';

  for (var k = 0; k < posts.length; k++) {
    var po = posts[k];
    h += '<div class="post-card">' +
      '<div class="post-header">' +
        '<div class="post-left">' +
          '<span class="post-id">#' + esc(po.cleanId) + '</span>' +
          '<span class="post-title">' + esc(po.title) + '</span>' +
        '</div>' +
        '<div class="post-meta">' + esc(po.author) + ' · ' + esc(po.time) + '</div>' +
      '</div>' +
      '<div class="post-content">';

    if (po.heat) h += '<div class="post-heat">🔥 ' + esc(po.heat) + '</div>';
    h += '<div class="post-body">' + esc(po.body) + '</div>';

    if (po.reps && po.reps.length > 0) {
      h += '<div class="replies-wrap">';
      for (var r = 0; r < po.reps.length; r++) {
        var rp = po.reps[r];
        h += '<div class="reply-item">' +
          '<div class="reply-author"><span>' + esc(rp.author) + '</span><span style="font-size:11px;color:#64748b;font-weight:normal;">' + esc(rp.time) + '</span></div>' +
          '<div class="reply-text">' + esc(rp.text) + '</div>' +
        '</div>';
      }
      h += '</div>';
    }

    h += '</div></div>';
  }

  var box = document.getElementById('forumBox');
  box.innerHTML = h;

  box.addEventListener('click', function(e) {
    var header = e.target.closest('.post-header');
    if (header) {
      header.parentNode.classList.toggle('open');
    }
  });
}
renderForum();
</script>
