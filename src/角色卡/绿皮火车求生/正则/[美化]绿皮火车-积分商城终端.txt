(function() {
var cs = document.currentScript;
var root = cs ? cs.previousElementSibling : document.querySelector('.shop-wrap:last-of-type');
if (!root) return;
var rawElem = root.querySelector('.shop-raw');
var renderTarget = root.querySelector('.shop-render');
var raw = rawElem ? rawElem.value : '';
function esc(s) {
var a = String.fromCharCode(38);
return String(s || '').replace(/&/g, a + 'amp;').replace(/</g, a + 'lt;').replace(/>/g, a + 'gt;').replace(/"/g, a + 'quot;').replace(/'/g, a + '#39;');
}
window.sendTrainBuy = function(name) {
var cmd = '购买：' + name;
try { if (typeof triggerSlash === 'function') triggerSlash('/send ' + cmd + '|/trigger'); } catch(e) {}
};
var lines = raw.split('\n');
var rows = [];
for (var i = 0; i < lines.length; i++) {
var s = lines[i].trim();
if (s.indexOf('[') !== -1 && s.indexOf(']') !== -1) {
var content = s.substring(s.indexOf('[') + 1, s.lastIndexOf(']'));
var parts = content.split('|');
if (parts.length === 1) parts = content.split('｜');
for (var j = 0; j < parts.length; j++) parts[j] = parts[j].trim();
if (parts[0]) rows.push(parts);
}
}
var bal = '0', ref = '下站刷新';
for (var i = 0; i < rows.length; i++) {
if (rows[i][0] === '余额') bal = rows[i][1];
if (rows[i][0] === '刷新') ref = rows[i][1];
}
var h = '<div class="shop-top"><div><div class="shop-title">🛒 积分物资兑换终端</div><div class="shop-refresh">刷新节点: ' + esc(ref) + '</div></div><div class="shop-balance">🪙 余额: ' + esc(bal) + '</div></div><div class="goods-grid">';
for (var i = 0; i < rows.length; i++) {
var r = rows[i];
if (r[0] === '分类') {
h += '<div class="cate-label">' + esc(r[1]) + '</div>';
} else if (r[0] === '商品') {
var name = r[1] || '物资', price = r[2] || '0', stock = r[3] || '0', desc = r[4] || '';
var sold = (parseFloat(stock) === 0);
h += '<div class="goods-row"><div class="goods-info"><div class="goods-name">' + esc(name) + ' <span style="font-size:11px;color:#94a3b8;font-weight:normal;">(库存: ' + esc(stock) + ')</span></div><div class="goods-desc">' + esc(desc) + '</div></div><div class="goods-price">' + esc(price) + ' 分</div>';
if (sold) {
h += '<div class="btn-buy soldout">售罄</div>';
} else {
h += '<div class="btn-buy" onclick="window.sendTrainBuy(\'' + esc(name) + '\')">兑换</div>';
}
h += '</div>';
}
}
h += '</div>';
renderTarget.innerHTML = h;
})();
