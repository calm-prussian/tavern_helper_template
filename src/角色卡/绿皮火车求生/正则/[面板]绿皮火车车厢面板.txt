  function num(v, d) {
    var n = parseFloat(v);
    return isNaN(n) ? d : n;

  function esc(s) {
    var a = String.fromCharCode(38);
    return String(s || '').replace(/&/g, a + 'amp;').replace(/</g, a + 'lt;').replace(/>/g, a + 'gt;').replace(/"/g, a + 'quot;').replace(/'/g, a + '#39;');
  }

  function getTagBlock(text, name) {
    var r = new RegExp('<(?:' + name + ')[^>]*>([\\s\\S]*?)<\\/(?:' + name + ')>', 'i');
    var m = String(text || '').match(r);
    return m ? m[1] : '';
  }

  function getAllTagBlocks(text, name) {
    var out = [];
    var r = new RegExp('<(?:' + name + ')[^>]*>([\\s\\S]*?)<\\/(?:' + name + ')>', 'gi');
    var m;
    while ((m = r.exec(String(text || ''))) !== null) { out.push(m[1]); }
    return out;
  }

  function parseRows(str) {
    var out = [];
    var lines = String(str || '').split('\n');
    for (var i = 0; i < lines.length; i++) {
      var s = lines[i].replace(/^\s+|\s+$/g, '');
      var m = s.match(/\[([^\]]+)\]/);
      if (!m) continue;
      var parts = m[1].replace(/｜/g, '|').split('|');
      for (var j = 0; j < parts.length; j++) parts[j] = parts[j].replace(/^\s+|\s+$/g, '');
      if (parts[0]) out.push(parts);
    }
    return out;
  }

  function getV(rows, key, def) {
    for (var i = 0; i < rows.length; i++) {
      if (rows[i][0] === key && rows[i][1] !== undefined && rows[i][1] !== '') return rows[i][1];
    }
    return def || '';
  }

  function getAll(rows, key) {
    var out = [];
    for (var i = 0; i < rows.length; i++) {
      if (rows[i][0] === key) out.push(rows[i]);
    }
    return out;
  }

        oldData.user.items = itArr;

  function canonicalRoom(room) {
    var s = String(room || '');
    var m = s.match(/\d+/);
    if (m) {
      var n = parseInt(m[0], 10);
      if (n >= 1 && n <= 10) return n + '号房';
    }
    return s || '未知房';
  }

  function normalizeBeds(beds) {
    var out = {};
    Object.keys(beds || {}).forEach(function(key) {
      var label = canonicalRoom(key);
      var r = beds[key] || {};
      if (!out[label]) out[label] = { 左铺: '空', 右铺: '空' };
      var target = out[label];
      ['左铺', '右铺'].forEach(function(side) {
        var name = r[side];
        if (!name || name === '空') return;
        if (target[side] === name) return;
        if (target[side] === '空') { target[side] = name; return; }
        if (target.左铺 === name || target.右铺 === name) return;
      });
    });
    return out;
  }

  function normalizeSite(site) {

    if (!site) return defaults;
    if (typeof site === 'string') { defaults.剩余 = site; return defaults; }
    var out = {};
    Object.keys(defaults).forEach(function(k) {
      out[k] = site[k] !== undefined && site[k] !== null ? site[k] : defaults[k];
    });
    if (!Array.isArray(out.已探索)) out.已探索 = out.已探索 ? [String(out.已探索)] : [];
    return out;
  }

  function fmtBed(name) {

  function findBed(beds, name) {
    var rooms = Object.keys(beds || {});
    for (var i = 0; i < rooms.length; i++) {
      var r = beds[rooms[i]] || {};
      if (r.左铺 === name) return rooms[i] + ' · 左铺';
      if (r.右铺 === name) return rooms[i] + ' · 右铺';
    }
    return '';
  }

  function normalizeCat(c) {
    var s = String(c || '').replace(/^\s+|\s+$/g, '');
    if (!s) return '其他';
    if (STORE_ORDER.indexOf(s) !== -1) return s;
    if (s.indexOf('水果') !== -1) return '食物';
    if (s.indexOf('食') !== -1) return '食物';
    if (s.indexOf('饮') !== -1 || s === '水') return '饮水';
    if (s.indexOf('药') !== -1) return '药品';
    if (s.indexOf('装备') !== -1 || s.indexOf('武器') !== -1) return '装备';
    if (s.indexOf('材料') !== -1) return '材料';
    return '其他';
  }

  function importLegacy() {
    var raw = getRawContent();
    if (raw.indexOf('<状态栏>') === -1) {
      alert('当前楼层没有旧状态栏可导入。');
      return;
    }
    var id = getCurrentId();
    if (id === null) return;
    var data = buildStatFromLegacy(mergeState(loadPersistData(), raw));
    var newText = raw.replace(/<状态栏>[\s\S]*?<\/状态栏>/i, '').replace(/\s+$/, '');
    if (newText.indexOf('<StatusPlaceHolderImpl/>') === -1) {
      newText += '\n\n<StatusPlaceHolderImpl/>';
    }

    try {
      var tasks = [];
      if (typeof Mvu !== 'undefined' && Mvu && typeof Mvu.replaceMvuData === 'function') {
        var m = Mvu.getMvuData({ type: 'message', message_id: id }) || {};
        m.stat_data = data;
        tasks.push(Mvu.replaceMvuData(m, { type: 'message', message_id: id }));
      } else if (typeof getVariables === 'function' && typeof replaceVariables === 'function') {
        var vars = getVariables({ type: 'message', message_id: id }) || {};
        vars.stat_data = data;
        replaceVariables(vars, { type: 'message', message_id: id });
      } else {
        alert('当前环境不支持写入 MVU 变量。');
        return;
      }

      if (typeof setChatMessages === 'function') {
        tasks.push(setChatMessages([{ message_id: id, message: newText }], { refresh: 'affected' }));
      } else {
        alert('已写入 MVU 变量，但当前环境无法移除旧状态栏；下一层新消息会使用 MVU。');
      }

      if (tasks.length) {
        Promise.all(tasks)
          .then(function() { location.reload(); })
          .catch(function(e) {
            console.error(e);
            alert('导入失败: ' + e);
          });
      } else {
        location.reload();
      }
    } catch (e) {
      console.error(e);
      alert('导入失败: ' + e);
    }
  }
  window.importTrainLegacy = importLegacy;

  function bindTabs(app) {
    var tabs = app.querySelectorAll('.m-tab');
    tabs.forEach(function(tb) {
      tb.addEventListener('click', function() {
        tabs.forEach(function(t) { t.classList.remove('active'); });
        app.querySelectorAll('.tab-pane').forEach(function(p) { p.classList.remove('active'); });
        tb.classList.add('active');
        var target = app.querySelector('#' + tb.getAttribute('data-tab'));
        if (target) target.classList.add('active');
      });
    });

    var cBtns = app.querySelectorAll('.char-btn');
    cBtns.forEach(function(btn) {
      btn.addEventListener('click', function() {
        cBtns.forEach(function(b) { b.classList.remove('active'); });
        app.querySelectorAll('.char-card-body').forEach(function(cb) { cb.classList.remove('active'); });
        btn.classList.add('active');
        var bodyTarget = app.querySelector('#charBody_' + btn.getAttribute('data-cidx'));
        if (bodyTarget) bodyTarget.classList.add('active');
      });
    });
  }

  function init() {
    var raw = getRawContent();
    var hasLegacy = raw.indexOf('<状态栏>') !== -1;
    var stat = hasLegacy ? null : getStatData();
    var finalState = stat ? stateFromStat(stat) : mergeState(loadPersistData(), raw);
    renderPanel(finalState, stat ? 'MVU' : '旧文本', !stat && hasLegacy);

    if (!stat && !hasLegacy) {
      var tries = 0;
      var timer = setInterval(function() {
        tries++;
        var latest = getStatData();
        if (latest) {
          clearInterval(timer);
          renderPanel(stateFromStat(latest), 'MVU', false);
        } else if (tries >= 20) {
          clearInterval(timer);
        }
      }, 500);
    }
