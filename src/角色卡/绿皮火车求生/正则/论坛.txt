<script>
(function() {
var cs = document.currentScript;
var root = cs ? cs.previousElementSibling : document.querySelector('.forum-wrap:last-of-type');
if (!root) return;
var rawElem = root.querySelector('.forum-raw');
var container = root.querySelector('.forum-render');
var raw = rawElem ? rawElem.value : '';
function esc(s) {
var a = String.fromCharCode(38);
return String(s || '').replace(/&/g, a + 'amp;').replace(/</g, a + 'lt;').replace(/>/g, a + 'gt;').replace(/"/g, a + 'quot;').replace(/'/g, a + '#39;');
}
var lines = raw.split('\n');
var rows = [], posts = [], map = {};
for (var i = 0; i < lines.length; i++) {
var s = lines[i].trim();
if (s.indexOf('[') !== -1 && s.indexOf(']') !== -1) {
var content = s.substring(s.indexOf('[') + 1, s.lastIndexOf(']'));
var parts = content.split('|');
if (parts.length === 1) parts = content.split('｜');
for (var j = 0; j < parts.length; j++) parts[j] = parts[j].trim();
if (parts[0]) rows.push(parts);
}
}
for (var i = 0; i < rows.length; i++) {
var r = rows[i];
if (r[0] === '帖') {
var p = { id: r[1] || '?', author: r[2] || '匿名', time: r[3] || '', title: r[4] || '无题', body: r[7] || '', reps: [] };
posts.push(p);
map[p.id] = p;
} else if (r[0] === '回复') {
var pid = r[1];
if (map[pid]) {
map[pid].reps.push({ author: r[2] || '匿名', time: r[3] || '', text: r[4] || '' });
}
}
}
if (!posts.length) {
container.innerHTML = '<div style="text-align:center;color:#64748b;font-size:12px;padding:16px;">当前频段未截获新帖子…</div>';
return;
}
var h = '';
for (var k = 0; k < posts.length; k++) {
var po = posts[k];
h += '<div class="post-item' + (k === 0 ? ' open' : '') + '">';
h += '<div class="post-head" onclick="this.parentNode.classList.toggle(\'open\')">';
h += '<div class="post-title"><span class="tag-id">#' + esc(po.id) + '</span>' + esc(po.title) + '</div>';
h += '<div class="post-meta">' + esc(po.author) + (po.time ? ' · ' + esc(po.time) : '') + '</div></div>';
h += '<div class="post-body">' + esc(po.body);
if (po.reps && po.reps.length) {
h += '<div class="reply-list">';
for (var m = 0; m < po.reps.length; m++) {
var rp = po.reps[m];
h += '<div class="reply-item"><span class="reply-user">' + esc(rp.author) + ':</span><span class="reply-text">' + esc(rp.text) + '</span>' + (rp.time ? '<span class="reply-time">' + esc(rp.time) + '</span>' : '') + '</div>';
}
h += '</div>';
}
h += '</div></div>';
}
container.innerHTML = h;
})();
</script>
